    for(let i=0;i<bin.length;i++) out[i] = bin.charCodeAt(i);
    return out;
  }
  function bytesToHex(bytes){
    return Array.from(bytes).map(b => b.toString(16).padStart(2,"0")).join("");
  }

  async function deriveKey(passphrase, room){
    const enc = new TextEncoder();
    const baseKey = await crypto.subtle.importKey(
      "raw", enc.encode(passphrase), "PBKDF2", false, ["deriveKey"]
    );
    return crypto.subtle.deriveKey(
      { name:"PBKDF2", salt: enc.encode("cipher-salt::" + room.toLowerCase().trim()), iterations: 150000, hash: "SHA-256" },
      baseKey, { name:"AES-GCM", length:256 }, false, ["encrypt","decrypt"]
    );
  }

  async function computeFingerprint(passphrase, room){
    const enc = new TextEncoder();
    const digest = await crypto.subtle.digest("SHA-256", enc.encode("fingerprint::" + room.toLowerCase().trim() + "::" + passphrase));
    const hex = bytesToHex(new Uint8Array(digest)).toUpperCase();
    return hex.slice(0,4) + "-" + hex.slice(4,8) + "-" + hex.slice(8,12);
  }

  async function encryptText(key, text){
    const iv = crypto.getRandomValues(new Uint8Array(12));
    const enc = new TextEncoder();
    const ctBuf = await crypto.subtle.encrypt({ name:"AES-GCM", iv }, key, enc.encode(text));
    return { iv: bytesToBase64(iv), ct: bytesToBase64(new Uint8Array(ctBuf)) };
  }
  async function decryptText(key, ivB64, ctB64){
    const iv = base64ToBytes(ivB64);
    const ct = base64ToBytes(ctB64);
    const ptBuf = await crypto.subtle.decrypt({ name:"AES-GCM", iv }, key, ct);
    return new TextDecoder().decode(ptBuf);
  }

  function friendlyAuthError(e){
    const code = e && e.code;
    switch(code){
      case "auth/email-already-in-use": return "An account with that email already exists — try logging in instead.";
      case "auth/invalid-email": return "That email address doesn't look right.";
      case "auth/weak-password": return "Password should be at least 8 characters.";
      case "auth/user-not-found":
      case "auth/wrong-password":
      case "auth/invalid-credential": return "Incorrect email or password.";
      case "auth/too-many-requests": return "Too many attempts — wait a bit and try again.";
      case "auth/missing-email": return "Enter your email first.";
      case "auth/invalid-action-code":
      case "auth/expired-action-code": return "This sign-in link is invalid or has expired — request a new one.";
      case "auth/invalid-phone-number": return "That phone number doesn't look right — include the country code.";
      case "auth/invalid-verification-code": return "That code doesn't match — check it and try again.";
      case "auth/code-expired": return "That code expired — request a new one.";
      case "auth/missing-phone-number": return "Enter a phone number first.";
      default: return "Something went wrong. Try again.";
    }
  }

  async function handleAuthSubmit(){
    const email = $("#in-auth-email").value.trim();
    const password = $("#in-auth-password").value;
    clearAuthMessages();

    if(!STORAGE_AVAILABLE){
      showAuthError("Couldn't connect to the backend. Check your internet connection and reload.");
      return;
    }
    if(!email || !password){
      showAuthError("Enter both an email and a password.");
      return;
    }

    const btn = $("#btn-auth-submit");
    btn.disabled = true;
    btn.textContent = "Working…";
    try{
      if(authMode === "login"){
        await auth.signInWithEmailAndPassword(email, password);
      }else{
        await auth.createUserWithEmailAndPassword(email, password);
      }
      $("#in-auth-password").value = "";
    }catch(e){
      showAuthError(friendlyAuthError(e));
    }finally{
      btn.disabled = false;
      btn.textContent = authMode === "login" ? "Log in" : "Create account";
    }
  }

  /* ================= ROOM STORAGE ================= */

  async function loadRoomMessages(room){
    const snap = await db.ref("rooms/" + roomKey(room) + "/messages").once("value");
    const val = snap.val();
    if(!val) return [];
    return Object.keys(val)
      .map(k => Object.assign({ id: k }, val[k]))
      .sort((a,b) => a.ts - b.ts);
  }
  async function appendRoomMessage(room, msgObj){
    const { id, ...rest } = msgObj;
    await db.ref("rooms/" + roomKey(room) + "/messages").push(rest);
  }
  async function purgeRoom(room){
    await db.ref("rooms/" + roomKey(room) + "/messages").remove();
  }

  function initials(name){
    return name.trim().split(/\s+/).slice(0,2).map(w => w[0].toUpperCase()).join("") || "?";
  }

  function renderEmptyIfNeeded(){
    if(thread.children.length === 0){
      const div = document.createElement("div");
      div.className = "empty-state";
      div.id = "empty-state";
      div.innerHTML = '<div class="ic"><svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M21 15a2 2 0 0 1-2 2H7l-4 4V5a2 2 0 0 1 2-2h14a2 2 0 0 1 2 2z"></path></svg></div><p>No messages yet. Anything sent here is only readable by people with this passphrase.</p>';
      thread.appendChild(div);
    }
  }
  function clearEmptyState(){
    const es = document.getElementById("empty-state");
    if(es) es.remove();
  }
  function formatTime(ts){
    return new Date(ts).toLocaleTimeString([], {hour:'2-digit', minute:'2-digit'});
  }

  function renderMessage(m, decryptedText, failed){
    clearEmptyState();
    const isMine = m.senderTag === senderTag;
    const row = document.createElement("div");
    row.className = "msg-row " + (isMine ? "mine" : "theirs") + (failed ? " failed" : "");
    if(!isMine){
      const sender = document.createElement("div");
      sender.className = "msg-sender";
      sender.textContent = m.sender || "unknown";
      row.appendChild(sender);
    }
    const bubble = document.createElement("div");
    bubble.className = "bubble";
    bubble.textContent = failed ? "Couldn't decrypt this message" : decryptedText;
    row.appendChild(bubble);
    const time = document.createElement("div");
    time.className = "msg-time";
    time.textContent = formatTime(m.ts);
    row.appendChild(time);
    thread.appendChild(row);
  }

  async function renderNewMessages(messages){
    let addedAny = false;
    for(const m of messages){
      if(knownIds.has(m.id)) continue;
      knownIds.add(m.id);
      addedAny = true;
      try{
        const text = await decryptText(cryptoKey, m.iv, m.ct);
        renderMessage(m, text, false);
      }catch(e){
        renderMessage(m, null, true);
      }
    }
    if(addedAny) thread.scrollTop = thread.scrollHeight;
    renderEmptyIfNeeded();
  }


  let currentMsgRef = null;

  async function handleSnapshot(snapshot){
    const val = snapshot.val();
    if(!val){
      if(knownIds.size > 0){
        knownIds.clear();
        thread.innerHTML = "";
      }
      renderEmptyIfNeeded();
      return;
    }
    const messages = Object.keys(val)
      .map(k => Object.assign({ id: k }, val[k]))
      .sort((a,b) => a.ts - b.ts);
    await renderNewMessages(messages);
  }

  function startListening(){
    currentMsgRef = db.ref("rooms/" + roomKey(roomId) + "/messages");
    currentMsgRef.on("value", handleSnapshot, () => {
      liveDot.classList.add("off");
      liveLabel.textContent = "sync error";
      setStatus("Lost connection — will retry", true);
    });
    db.ref(".info/connected").on("value", (snap) => {
      const connected = snap.val() === true;
      liveDot.classList.toggle("off", !connected);
      liveLabel.textContent = connected ? "live" : "connecting…";
    });
  }

  function stopListening(){
    if(currentMsgRef) currentMsgRef.off("value", handleSnapshot);
    currentMsgRef = null;
    db.ref(".info/connected").off();
  }

  async function verifyPassphrase(room, key){
    const messages = await loadRoomMessages(room);
    if(messages.length === 0) return true;
    try{
      await decryptText(key, messages[0].iv, messages[0].ct);
      return true;
    }catch(e){
      return false;
    }
  }

  async function enterRoom(){
    const room = $("#in-room").value.trim();
    const pass = $("#in-pass").value;
    const name = $("#in-name").value.trim();

    gateError.classList.remove("show");
    gateError.textContent = "";

    if(!STORAGE_AVAILABLE){
      gateError.textContent = "Couldn't connect to the messaging backend. Check your internet connection and reload.";
      gateError.classList.add("show");
      return;
    }
    if(!room || !pass || !name){
      gateError.textContent = "Fill in all three fields to continue.";
      gateError.classList.add("show");
      return;
    }
    if(pass.length < 6){
      gateError.textContent = "Passphrase should be at least 6 characters.";
      gateError.classList.add("show");
      return;
    }

    $("#btn-enter").disabled = true;
    $("#btn-enter").textContent = "Checking…";

    try{
      cryptoKey = await deriveKey(pass, room);

      const ok = await verifyPassphrase(room, cryptoKey);
      if(!ok){
        gateError.textContent = "That passphrase doesn't match this room ID. Check it and try again.";
        gateError.classList.add("show");
        cryptoKey = null;
        return;
      }

      roomId = room;
      callsign = name;
      senderTag = crypto.randomUUID().slice(0,8);
      knownIds = new Set();

      $("#hdr-room").textContent = room;
      $("#hdr-avatar").textContent = initials(name);
      $("#fp-code").textContent = await computeFingerprint(pass, room);

      gateScreen.classList.remove("active");
      chatScreen.classList.add("active");
      thread.innerHTML = "";
      renderEmptyIfNeeded();
      startListening();
    }catch(e){
      gateError.textContent = "Something went wrong deriving the key. Try reloading.";
      gateError.classList.add("show");
    }finally{
      $("#btn-enter").disabled = false;
      $("#btn-enter").textContent = "Enter room";
    }
  }

  async function sendMessage(){
    const input = $("#in-msg");
    const text = input.value.trim();
    if(!text || sending) return;
    sending = true;
    $("#btn-send").disabled = true;
    try{
      const { iv, ct } = await encryptText(cryptoKey, text);
      const msgObj = { id: crypto.randomUUID(), sender: callsign, senderTag, iv, ct, ts: Date.now() };
      await appendRoomMessage(roomId, msgObj);
      input.value = "";
      input.style.height = "auto";
    }catch(e){
      setStatus("Message failed to send", true);
    }finally{
      sending = false;
      $("#btn-send").disabled = false;
      input.focus();
    }
  }

  function leaveRoom(){
    stopListening();
    cryptoKey = null; roomId = null; callsign = null; knownIds = new Set();
    chatScreen.classList.remove("active");
    gateScreen.classList.add("active");
    $("#in-pass").value = "";
    thread.innerHTML = "";
  }

    liveLabel.textContent = "backend unavailable";
  }

  });

  $("#btn-send").addEventListener("click", sendMessage);
  $("#in-msg").addEventListener("keydown", (e) => {

  });

    this.style.height = "auto";
    this.style.height = Math.min(this.scrollHeight, 110) + "px";
  });

  $("#btn-leave").addEventListener("click", leaveRoom);

  const purgeOverlay = $("#purge-overlay");
  $("#btn-purge").addEventListener("click", () => purgeOverlay.classList.add("active"));
  $("#purge-cancel").addEventListener("click", () => purgeOverlay.classList.remove("active"));

    purgeOverlay.classList.remove("active");
    try{
      await purgeRoom(roomId);
      knownIds = new Set();
      thread.innerHTML = "";
      renderEmptyIfNeeded();
      setStatus("Room cleared");
    }catch(e){
      setStatus("Couldn't clear room", true);
    }
  });

})();
